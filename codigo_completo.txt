template <class TL>
class Lista
{
private:

    template <class TE>
    class Elemento
    {
    private:
        TE pInfo{};
        Elemento<TE>* pProx;

    public:
        Elemento() : pProx(nullptr) {}
        ~Elemento() = default;

        void incluir(const TE& p) { pInfo = p; }
        void setProx(Elemento<TE>* p) { pProx = p; }
        Elemento<TE>* getProx() const { return pProx; }
        TE getInfo() const { return pInfo; }
    };

    Elemento<TL>* pPrimeiro;
    Elemento<TL>* pUltimo;
    int quantidade;

    // Impede cÛpia superficial da lista.
    Lista(const Lista&) = delete;
    Lista& operator=(const Lista&) = delete;

public:

    Lista() : pPrimeiro(nullptr), pUltimo(nullptr), quantidade(0) {}

    ~Lista()
    {
        limpar();
    }

    void incluir(const TL& p)
    {
        Elemento<TL>* novo = new Elemento<TL>();
        novo->incluir(p);

        if(pPrimeiro == nullptr)
        {
            pPrimeiro = novo;
            pUltimo = novo;
        }
        else
        {
            pUltimo->setProx(novo);
            pUltimo = novo;
        }
        ++quantidade;
    }

    void limpar()
    {
        Elemento<TL>* atual = pPrimeiro;

        while(atual != nullptr)
        {
            Elemento<TL>* proximo = atual->getProx();
            delete atual;
            atual = proximo;
        }

        pPrimeiro = nullptr;
        pUltimo = nullptr;
        quantidade = 0;
    }

    int tamanho() const { return quantidade; }

    template <class Funcao>
    void percorrer(Funcao acao) const
    {
        Elemento<TL>* atual = pPrimeiro;

        while(atual != nullptr)
        {
            acao(atual->getInfo());
            atual = atual->getProx();
        }
    }

    bool vazia() const
    {
        return pPrimeiro == nullptr;
    }
};

class ListaEntidades
{

    Lista<Entidade*> LEs;

public:

    ListaEntidades& operator=(const ListaEntidades&) = delete;

    void incluir(Entidade* p)
    {
        if(p != nullptr)
            LEs.incluir(p);
    }

    void percorrer()
    {
        LEs.percorrer([](Entidade* p)
        {
            if(p != nullptr)
                p->executar();
        });
    }

    void limpar()
    {
        LEs.limpar();
    }

    bool vazia() const
    {
        return LEs.vazia();
    }
};

Boitata::Boitata()
    : velocidadePerseguicao(1.5f),
      tempoRecargaFogo(120), // 120 quadros: aproximadamente 2 segundos a 60 FPS
      contadorTempoFogo(0),
      pFogo(new Projetil_Acido())

{
    posX = 600.f;
    posY = 200.f;
    x = posX;
    y = posY;
    vidas = 100;
    danoContato = 5;
    pontosAoMorrer = 1000;

    retangulo->setSize({97.f, 60.f});
    retangulo->setFillColor(sf::Color(190, 45, 35));
    carregarTextura("assets/Inimigos/Boitata.png");
    retangulo->setPosition({posX, posY});

    // Reutiliza a entidade de proj√©til do projeto, mas d√° a ela apar√™ncia de fogo.
    pFogo->getForma()->setFillColor(sf::Color(255, 105, 20));
}

Boitata::~Boitata()
{
    delete pFogo;
    pFogo = nullptr;
}

void Boitata::executar()
{
    // O proj√©til precisa continuar andando mesmo quando o Boitat√° n√£o tem alvo.
    if (pFogo != nullptr)
        pFogo->executar();

    if (!estaVivo() || alvo == nullptr)
        return;

    mover();

    // S√≥ dispara novamente quando o proj√©til anterior j√° saiu da tela.
    if (pFogo->estaAtivo())
        return;

    ++contadorTempoFogo;
    if (contadorTempoFogo < tempoRecargaFogo)
        return;

    const sf::Vector2f posAlvo = alvo->getForma()->getPosition();
    const sf::Vector2f tamanhoAlvo = alvo->getForma()->getSize();
    const float centroAlvo = posAlvo.x + tamanhoAlvo.x / 2.f;
    const float centroBoitata = posX + retangulo->getSize().x / 2.f;
    const float direcao = (centroAlvo >= centroBoitata) ? 1.f : -1.f;

    const float larguraProjetil = pFogo->getForma()->getSize().x;
    const float inicioX = (direcao > 0.f)
        ? posX + retangulo->getSize().x
        : posX - larguraProjetil;
    const float inicioY = posY + retangulo->getSize().y / 2.f;

    pFogo->ativar(inicioX, inicioY, direcao);
    contadorTempoFogo = 0;
}

void Boitata::mover()
{
    if (alvo == nullptr)
        return;

    const sf::Vector2f posAlvo = alvo->getForma()->getPosition();
    const sf::Vector2f tamanhoAlvo = alvo->getForma()->getSize();
    const float centroAlvoX = posAlvo.x + tamanhoAlvo.x / 2.f;
    const float centroAlvoY = posAlvo.y + tamanhoAlvo.y / 2.f;

    const float centroBoitataX = posX + retangulo->getSize().x / 2.f;
    const float centroBoitataY = posY + retangulo->getSize().y / 2.f;

    // Movimento horizontal (eixo X)
    if (centroAlvoX > centroBoitataX)
        posX += velocidadePerseguicao;
    else if (centroAlvoX < centroBoitataX)
        posX -= velocidadePerseguicao;

    // Movimento vertical (eixo Y)
    const float velocidadeVertical = velocidadePerseguicao * 0.8f;
    if (centroAlvoY > centroBoitataY)
        posY += velocidadeVertical;
    else if (centroAlvoY < centroBoitataY)
        posY -= velocidadeVertical;

    // Limites horizontais da arena (paredes em x=20 e x=780)
    if (posX < 20.f)
        posX = 20.f;
    if (posX + retangulo->getSize().x > 780.f)
        posX = 780.f - retangulo->getSize().x;

    // Limites verticais da arena (teto em y=20 e ch√£o em y=560)
    if (posY < 20.f)
        posY = 20.f;
    if (posY + retangulo->getSize().y > 560.f)
        posY = 560.f - retangulo->getSize().y;

    x = posX;
    y = posY;
    retangulo->setPosition({posX, posY});
}

void Boitata::salvar()
{
    x = posX;
    y = posY;

    salvarDataBuffer();

    Salvamento << " " << vidas
               << " " << contadorTempoFogo;

    pFogo->salvar();
    Salvamento << " " << pFogo->getDadosSalvos();
}

void Boitata::danificar(Jogador* pJogador)
{
    if (pJogador != nullptr)
    {
        pJogador->receberDano(danoContato);

        const float centroBoitata = posX + retangulo->getSize().x / 2.f;
        const float centroJogador = pJogador->getX() + pJogador->getForma()->getSize().x / 2.f;
        const float direcao = (centroJogador >= centroBoitata) ? 1.f : -1.f;

        pJogador->arremessar(direcao);
    }
}

Projetil_Acido* Boitata::getProjetil() const
{
    return pFogo;
}

void Boitata::setEstado(int v, int cont)
{
    vidas = v;
    contadorTempoFogo = cont;
}

    : direcao(-1),
      velocidadeAndar(2.5f),
      limiteEsquerdo(25.f),
      limiteDireito(775.f)
{
    posX = 600.f;
    posY = 510.f;
    x = posX;
    y = posY;
    vidas = 1;
    danoContato = 1;
    sf::Color marrom(139, 69, 19);

    retangulo->setSize({44.f, 50.f});
    retangulo->setFillColor(marrom);
    carregarTextura("assets/Inimigos/Cogumelo.png");
    retangulo->setPosition({posX, posY});
}

Capanga_Cogumelo::~Capanga_Cogumelo() {}

void Capanga_Cogumelo::executar() {
    if (!estaVivo())
        return;
    mover();
}

void Capanga_Cogumelo::inverterDirecao() {
    direcao *= -1;
}

void Capanga_Cogumelo::definirLimitesPatrulha(float esq, float dir) {
    limiteEsquerdo = esq;
    limiteDireito = dir;
}

void Capanga_Cogumelo::setEstado(int v, int dir) {
    vidas = v;
    direcao = dir;
}

void Capanga_Cogumelo::danificar(Jogador* pJogador) {
    pJogador->receberDano(danoContato);

    const float centroCogumelo = posX + retangulo->getSize().x / 2.f;
    const float centroJogador = pJogador->getX() + pJogador->getForma()->getSize().x / 2.f;
    const float dirEmpurrao = (centroJogador >= centroCogumelo) ? 1.f : -1.f;
    pJogador->arremessar(dirEmpurrao);

    inverterDirecao();
}

void Capanga_Cogumelo::mover() {
    posX += velocidadeAndar * direcao;
    const float largura = retangulo->getSize().x;

    if (posX <= limiteEsquerdo) {
        posX = limiteEsquerdo;
        direcao = 1;
    } else if (posX + largura >= limiteDireito) {
        posX = limiteDireito - largura;
        direcao = -1;
    }

    x = posX;
    y = posY;
    retangulo->setPosition({posX, posY});
}

void Capanga_Cogumelo::salvar() {
    x = posX;
    y = posY;

    salvarDataBuffer();

    Salvamento << " " << vidas
               << " " << direcao
               << " " << limiteEsquerdo
               << " " << limiteDireito;
}

bool Capanga_Cogumelo::estaVivo() {
    return vidas > 0;
}

std::mt19937& Fase::geradorAleatorio()
{
    static std::mt19937 gerador{std::random_device{}()};
    return gerador;
}

int Fase::sortear(int minimo, int maximo)
{
    return std::uniform_int_distribution<int>(minimo, maximo)(geradorAleatorio());
}

void Fase::espalharCapangas(int quantidade, const list<Plataforma*>& plats)
{
    const float larguraCogumelo = 44.f;
    const float alturaCogumelo  = 50.f;

    for (int i = 0; i < quantidade; ++i)
    {
        // 50% de chance de ir em plataforma (se houver), 50% de ir no ch„o
        bool emPlataforma = !plats.empty() && (sortear(0, 1) == 1);

        if (emPlataforma)
        {
            int idx = sortear(0, static_cast<int>(plats.size()) - 1);
            Plataforma* p = *std::next(plats.begin(), idx);
            sf::FloatRect b = p->getForma()->getGlobalBounds();

            float xMin = b.position.x + 4.f;
            float xMax = b.position.x + b.size.x - larguraCogumelo - 4.f;
            float xSpawn = (xMax > xMin)
                ? static_cast<float>(sortear(static_cast<int>(xMin), static_cast<int>(xMax)))
                : b.position.x + (b.size.x - larguraCogumelo) / 2.f;

            criarCapanga(xSpawn, b.position.y - alturaCogumelo,
                         b.position.x, b.position.x + b.size.x);
        }
        else
        {
            // Espalhado pelo ch„o (longe do spawn inicial do herÛi em x=100..180)
            float xSpawn = static_cast<float>(sortear(200, 720));
            criarCapanga(xSpawn, 560.f - alturaCogumelo, 25.f, 775.f);
        }
    }
}

void Fase::criarPlataforma(float x, float y, int largura, int altura,
                           Obstaculo* plataforma)
{
    if (numObstaculos >= maxObstaculos)
        return;

    plataforma->getForma()->setSize({
        static_cast<float>(largura),
        static_cast<float>(altura)
    });
    plataforma->setPosicao(x, y);

    ++numObstaculos;
    pFisica->incluirObstaculo(plataforma);
    listaEntidades.incluir(plataforma);
}

bool Fase::registrarInimigo(CriaturaHostil* inimigo)
{
    if (inimigo == nullptr || numInimigos >= maxInimigos)
        return false;

    ++numInimigos;
    pFisica->incluirInimigos(inimigo);
    listaEntidades.incluir(inimigo);

    return true;
}

list<Plataforma*> Fase::gerarPlataformasAleatorias(int quantidade)
{
    auto sorteia = [&](int a, int b) { return sortear(a, b); };

    const int andares[3] = {425, 290, 155};
    const int espessura  = 20;
    const int xMin = 40, xMax = 760;
    const int folga   = 60;
    const int alcance = 110;

    struct Dados { int andar, x, largura; };
    list<Dados> feitas;
    list<Plataforma*> criadas;
    int maiorAndar = -1;

    for (int tentativa = 0; tentativa < 300 && static_cast<int>(criadas.size()) < quantidade; ++tentativa)
    {
        if (numObstaculos >= maxObstaculos)
            break;

        int andar   = sorteia(0, std::min(maiorAndar + 1, 2));
        int largura = sorteia(110, 200);
        int x       = sorteia(xMin, xMax - largura);
        int y       = andares[andar] + sorteia(-8, 8);

        bool ok = true;
        bool temApoio = (andar == 0);
        for (const Dados& d : feitas)
        {
            int vao = std::max(x - (d.x + d.largura), d.x - (x + largura));
            if (d.andar == andar && vao < folga)        { ok = false; break; }
            if (d.andar == andar - 1 && vao <= alcance) temApoio = true;
        }
        if (!ok || !temApoio)
            continue;

        Plataforma* p = new Plataforma(x, y, largura, espessura);
        criarPlataforma(x, y, largura, espessura, p);
        criadas.push_back(p);
        feitas.push_back({andar, x, largura});
        maiorAndar = std::max(maiorAndar, andar);
    }
    return criadas;
}

void Fase::salvar(const std::string& pasta)
{
    std::filesystem::create_directories(pasta);

    // 4. fase.txt (metadados da fase atual)
    std::ofstream arqFase(pasta + "fase.txt");
    if (arqFase.is_open())
    {
        arqFase << numeroFase << " " << concluida << "\n";
    }

    if (pFisica != nullptr)
        pFisica->salvarArquivos(pasta);
}

bool Fase::carregarArquivos(const std::string& pasta)
{
    std::ifstream arqJog(pasta + "jogador.txt");
    std::ifstream arqObs(pasta + "obstaculos.txt");
    std::ifstream arqIni(pasta + "inimigos.txt");

    if (!arqJog.is_open() || !arqObs.is_open() || !arqIni.is_open())
        return false;

    // Limpa o estado atual da fase antes de recriar tudo do arquivo
    listaEntidades.limpar();
    if (pFisica != nullptr)
        pFisica->destruirEntidades();

    numInimigos = 0;
    numObstaculos = 0;

    // 1. Carrega o Jogador (ordem salva em Jogador::salvar)
    if (pJogador != nullptr)
    {
        float jx, jy, vx, vy;
        int vidas, pontos;

        if (arqJog >> jx >> jy >> vidas >> pontos >> vx >> vy)
        {
            pJogador->setX(jx);
            pJogador->setY(jy);
            pJogador->setEstado(vx, vy, vidas, pontos);

            pFisica->getJogador(pJogador);
            listaEntidades.incluir(pJogador);
        }
    }

    // 2. Carrega os Obst·culos (limites da arena, plataformas, lamas e lavas)
    std::string tipo;
    while (arqObs >> tipo)
    {
        float ox, oy;
        bool dano;
        arqObs >> ox >> oy >> dano;

        if (tipo == "PLATAFORMA")
        {
            int larg, alt, dir;
            bool movel;
            float yBase, amp, vel;
            arqObs >> larg >> alt >> movel >> yBase >> amp >> vel >> dir;

            Plataforma* p = new Plataforma(static_cast<int>(ox), static_cast<int>(oy), larg, alt);
            if (movel)
                p->configurarMovel(movel, yBase, amp, vel, dir);

            pFisica->incluirObstaculo(p);
            listaEntidades.incluir(p);
        }
        else if (tipo == "LAMA")
        {
            float larg, alt;
            arqObs >> larg >> alt;

            Lama* l = new Lama();
            l->getForma()->setSize({larg, alt});
            l->setPosicao(ox, oy);

            pFisica->incluirObstaculo(l);
            listaEntidades.incluir(l);
        }
        else if (tipo == "LAVA")
        {
            int danoLava;
            float larg, alt;
            arqObs >> danoLava >> larg >> alt;

            Lava* lv = new Lava();
            lv->getForma()->setSize({larg, alt});
            lv->setPosicao(ox, oy);

            pFisica->incluirObstaculo(lv);
            listaEntidades.incluir(lv);
        }
    }

    // 3. Carrega os Inimigos e o estado de seus projÈteis
    while (arqIni >> tipo)
    {
        float ix, iy;
        int vidas;
        arqIni >> ix >> iy >> vidas;

        if (tipo == "CAPANGA")
        {
            int dir;
            float limEsq, limDir;
            arqIni >> dir >> limEsq >> limDir;

            Capanga_Cogumelo* c = new Capanga_Cogumelo();
            c->setPosicao(ix, iy);
            c->setEstado(vidas, dir);
            c->definirLimitesPatrulha(limEsq, limDir);
            registrarInimigo(c);
        }
        else if (tipo == "BOITATA" || tipo == "PLANTA")
        {
            int contTempo, projDano;
            float px, py, pvx;
            bool projAtivo;
            arqIni >> contTempo >> px >> py >> pvx >> projDano >> projAtivo;

            if (tipo == "BOITATA")
            {
                Boitata* b = new Boitata();
                b->setPosicao(ix, iy);
                b->setEstado(vidas, contTempo);
                b->definirAlvo(pJogador);
                b->getProjetil()->restaurar(px, py, pvx, projAtivo);

                registrarInimigo(b);
                pFisica->incluirProjetil(b->getProjetil());
            }
            else
            {
                Planta_Acida* pl = new Planta_Acida();
                pl->setPosicao(ix, iy);
                pl->setEstado(vidas, contTempo);
                pl->definirAlvo(pJogador);
                pl->getProjetil()->restaurar(px, py, pvx, projAtivo);

                registrarInimigo(pl);
                pFisica->incluirProjetil(pl->getProjetil());
            }
        }
    }

    return true;
}

void Fase::tornarPlataformasMoveis(const list<Plataforma*>& plats,
                                   const list<Plataforma*>& fixas)
{
    const float alturaHeroi = (pJogador != nullptr)
        ? pJogador->getForma()->getSize().y : 40.f;
    const float folgaMinima = alturaHeroi + 15.f;
    const float amplitudeMax = 40.f;

    for (Plataforma* p : plats)
    {
        if (std::find(fixas.begin(), fixas.end(), p) != fixas.end())
            continue;

        sf::FloatRect a = p->getForma()->getGlobalBounds();
        float baixoAcima = 20.f;

        for (Obstaculo* o : pFisica->getObstaculos())
        {
            Plataforma* q = dynamic_cast<Plataforma*>(o);
            if (q == nullptr || q == p) continue;

            sf::FloatRect b = q->getForma()->getGlobalBounds();
            bool sobrepoeX = a.position.x < b.position.x + b.size.x &&
                             b.position.x < a.position.x + a.size.x;
            float baixoQ = b.position.y + b.size.y;

            if (sobrepoeX && baixoQ <= a.position.y)
                baixoAcima = std::max(baixoAcima, baixoQ);
        }

        float folgaAcima = a.position.y - baixoAcima;
        float amplitude  = std::min(amplitudeMax, folgaAcima - folgaMinima);
        if (amplitude < 15.f)
            continue;

        float velocidade = sortear(5, 12) / 10.f;
        p->tornarMovel(amplitude, velocidade);
    }
}

Fase_Boss::Fase_Boss(Jogador* jogador)
    : Fase(2, jogador, 5, 10) {} // 5 inimigos (4 cogumelos + 1 Boitat√°), at√© 10 obst√°culos

Fase_Boss::~Fase_Boss() {}

void Fase_Boss::iniciarFase() {
    std::cout << "[FASE 2] Iniciando a Caverna do Boitata...\n";

    if (pJogador != nullptr) {
        pJogador->setX(100.0f);
        pJogador->setY(500.0f);
        pFisica->getJogador(pJogador);
        listaEntidades.incluir(pJogador);
    }

    criarLimitesArena();

    // M√≠nimo 3 plataformas e m√≠nimo 3 / m√°ximo 5 lavas
    int numPlats = sortear(3, 5);
    int numLavas = sortear(3, 5);

    list<Plataforma*> plats = gerarPlataformasAleatorias(numPlats);

    // Embaralha plataformas para sortear onde colocar lavas a√©reas
    list<Plataforma*> embaralhadas;
    while (!plats.empty()) {
        int idx = sortear(0, static_cast<int>(plats.size()) - 1);
        auto it = std::next(plats.begin(), idx);
        embaralhadas.push_back(*it);
        plats.erase(it);
    }
    plats = embaralhadas;

    list<Plataforma*> fixas;
    auto itPlat = plats.begin();
    int livreChaoX = 200; // longe do spawn do jogador (x 100..160)

    for (int i = 0; i < numLavas; ++i) {
        // Decide se a lava vai em uma plataforma ou no ch√£o
        bool naPlataforma = (itPlat != plats.end()) && (sortear(0, 1) == 1);

        if (naPlataforma) {
            Plataforma* platAtual = *itPlat++;
            fixas.push_back(platAtual); // plataforma com lava fica parada

            sf::FloatRect b = platAtual->getForma()->getGlobalBounds();
            int larguraLava = std::min(80, static_cast<int>(b.size.x) - 20);
            int xLava = static_cast<int>(b.position.x + (b.size.x - larguraLava) / 2.f);

            criarPlataforma(static_cast<float>(xLava), b.position.y - 12.f,
                            larguraLava, 12, new Lava());
        } else {
            int larguraLava = sortear(55, 85);
            int xMax = std::max(livreChaoX, 740 - larguraLava);
            int xLava = sortear(livreChaoX, xMax);

            criarPlataforma(static_cast<float>(xLava), 548.f,
                            larguraLava, 12, new Lava());

            livreChaoX = std::min(680, xLava + larguraLava + 45);
        }
    }

    // Plataformas sem lava podem se mover
    tornarPlataformasMoveis(plats, fixas);

    // 4 Cogumelos espalhados entre ch√£o e plataformas
    espalharCapangas(4, plats);

    // Chef√£o Boitat√°
    Boitata* boss = new Boitata();
    boss->setPosicao(600.0f, 200.0f);
    boss->definirAlvo(pJogador);

    if (!registrarInimigo(boss)) {
        delete boss;
        return;
    }
    pFisica->incluirProjetil(boss->getProjetil());
}

void Fase_Boss::executar() {
    if (pJogador == nullptr || !pJogador->estaVivo()) return;

    listaEntidades.percorrer();
    pFisica->executar();
    desenhar();

    if (todosInimigosMortos())
        concluida = true;
}

void Fase_Boss::carregar() {
    numeroFase = 2;
    carregarArquivos("saves/");
}

    for (int i = 0; i < numLamas; ++i) {
        // Decide se a lama vai na plataforma (enquanto houver livre) ou no ch√£o
        bool colocarNoChao = (itPlat == plats.end()) || (sortear(0, 1) == 1);

        if (!colocarNoChao) {
            Plataforma* platAtual = *itPlat++;
            fixas.push_back(platAtual);

            sf::FloatRect b = platAtual->getForma()->getGlobalBounds();
            int larguraLama = std::max(30, static_cast<int>(b.size.x) - 20);
            int xLama = static_cast<int>(b.position.x) + 10;

            criarPlataforma(static_cast<float>(xLama), b.position.y - 10.f,
                            larguraLama, 10, new Lama());
        } else {
            // Lama no ch√£o (topo do ch√£o = y 560)
            int larguraLama = sortear(60, 100);
            int xMax = std::max(livreChaoX, 740 - larguraLama);
            int xLama = sortear(livreChaoX, xMax);

            criarPlataforma(static_cast<float>(xLama), 550.f,
                            larguraLama, 10, new Lama());

            livreChaoX = std::min(680, xLama + larguraLama + 40);
        }
    }

    // 3 Cogumelos espalhados entre ch√£o e plataformas
    espalharCapangas(3, plats);

    // 2 Plantas √Åcidas (completando os 5 inimigos da Fase 1)
    Planta_Acida* planta1 = new Planta_Acida();
    planta1->setPosicao(680.0f, 510.0f);
    planta1->definirAlvo(pJogador);

    Planta_Acida* planta2 = new Planta_Acida();
    planta2->definirAlvo(pJogador);
    if (!plats.empty()) {
        Plataforma* ultima = plats.back();
        if (std::find(fixas.begin(), fixas.end(), ultima) == fixas.end())
            fixas.push_back(ultima);

        sf::FloatRect b  = ultima->getForma()->getGlobalBounds();
        sf::Vector2f tam = planta2->getForma()->getSize();

        // Se a plataforma j√° tiver lama, coloca a planta no canto direito; sen√£o, centraliza
        float xPlanta = b.position.x + b.size.x - tam.x - 5.f;
        planta2->setPosicao(xPlanta, b.position.y - tam.y);
    }

    tornarPlataformasMoveis(plats, fixas);

    if (!registrarInimigo(planta1)) delete planta1;
    else pFisica->incluirProjetil(planta1->getProjetil());

    if (!registrarInimigo(planta2)) delete planta2;
    else pFisica->incluirProjetil(planta2->getProjetil());
}

    if (pJogador == nullptr || !pJogador->estaVivo()) return;

    listaEntidades.percorrer();
    pFisica->executar();
    desenhar();

    if (todosInimigosMortos())
        concluida = true;
}

    carregarArquivos("saves/");

void Gerenciador_CL::colisaoinimigo()
{
    if (pJog1 == nullptr)
        return;

    for (CriaturaHostil* pInimigo : LIs)
    {
        if (pInimigo == nullptr || !pInimigo->estaVivo())
            continue;

        if (!verificaColi(pJog1, pInimigo))
            continue;

        sf::FloatRect jogador = pJog1->getForma()->getGlobalBounds();
        sf::FloatRect inimigo = pInimigo->getForma()->getGlobalBounds();

        const bool sobrepoeX =
            jogador.position.x + jogador.size.x > inimigo.position.x &&
            jogador.position.x < inimigo.position.x + inimigo.size.x;

        const bool caiuSobre =
            pJog1->getVelocidadeY() > 0.f &&
            pJog1->getYAnterior() + jogador.size.y <= inimigo.position.y &&
            jogador.position.y + jogador.size.y >= inimigo.position.y &&
            sobrepoeX;

        if (caiuSobre)
        {
            pJog1->setY(inimigo.position.y - jogador.size.y);
            pInimigo->receberDano(2);
            pJog1->quicar();

            if (!pInimigo->estaVivo())
                pJog1->adicionarPontos(pInimigo->getPontos());
        }
        else
        {
            pInimigo->danificar(pJog1);
        }
    }
}

void Gerenciador_CL::colisaoProjetil()
{
    if (pJog1 == nullptr || !pJog1->estaVivo())
        return;

    for (Projetil_Acido* pProjetil : LPs)
    {
        if (pProjetil == nullptr || !pProjetil->estaAtivo())
            continue;

        if (verificaColi(pJog1, pProjetil))
        {
            pJog1->receberDano(pProjetil->getDano());
            pProjetil->desativar();
        }
    }
}

        }

            else if (dynamic_cast<Plataforma*>(o)) arqObs << "PLATAFORMA ";

            arqObs << o->getDadosSalvos() << "\n";
        }
    }

    // 3. inimigos.txt
    std::ofstream arqIni(pasta + "inimigos.txt");

void Gerenciador_CL::colisaoInimigoObstaculo()
{
    for (CriaturaHostil* pInimigo : LIs)
    {
        if (pInimigo == nullptr || !pInimigo->estaVivo())
            continue;

        const bool ehBoitata = (dynamic_cast<Boitata*>(pInimigo) != nullptr);

        for (Obstaculo* pObstaculo : LOs)
        {
            // Apenas Plataforma È sÛlida (Lama e Lava n„o bloqueiam inimigos)
            Plataforma* pPlat = dynamic_cast<Plataforma*>(pObstaculo);
            if (pPlat == nullptr)
                continue;

            sf::FloatRect ini = pInimigo->getForma()->getGlobalBounds();
            sf::FloatRect plat = pPlat->getForma()->getGlobalBounds();

            const float iniEsq = ini.position.x;
            const float iniDir = ini.position.x + ini.size.x;
            const float iniTopo = ini.position.y;
            const float iniBase = ini.position.y + ini.size.y;

            const float platEsq = plat.position.x;
            const float platDir = plat.position.x + plat.size.x;
            const float platTopo = plat.position.y;
            const float platBase = plat.position.y + plat.size.y;

            const bool sobrepoeX = (iniDir > platEsq) && (iniEsq < platDir);

            // Se cogumelo/planta estiver apoiado em uma plataforma mÛvel descendo, acompanha ela
            if (!ehBoitata && sobrepoeX && std::abs(iniBase - platTopo) <= 3.f)
            {
                pInimigo->setPosicao(pInimigo->getX(), platTopo - ini.size.y);
                continue;
            }

            if (!verificaColi(pInimigo, pPlat))
                continue;

            // Calcula o quanto penetrou em cada eixo para empurrar pelo caminho mais curto
            const float sobreposicaoX = std::min(iniDir, platDir) - std::max(iniEsq, platEsq);
            const float sobreposicaoY = std::min(iniBase, platBase) - std::max(iniTopo, platTopo);

            const float centroIniX = iniEsq + ini.size.x / 2.f;
            const float centroPlatX = platEsq + plat.size.x / 2.f;
            const float centroIniY = iniTopo + ini.size.y / 2.f;
            const float centroPlatY = platTopo + plat.size.y / 2.f;

            if (sobreposicaoY < sobreposicaoX)
            {
                if (centroIniY < centroPlatY)
                    pInimigo->setPosicao(pInimigo->getX(), platTopo - ini.size.y);
                else
                    pInimigo->setPosicao(pInimigo->getX(), platBase);
            }
            else
            {
                if (centroIniX < centroPlatX)
                    pInimigo->setPosicao(platEsq - ini.size.x, pInimigo->getY());
                else
                    pInimigo->setPosicao(platDir, pInimigo->getY());

                if (Capanga_Cogumelo* cap = dynamic_cast<Capanga_Cogumelo*>(pInimigo))
                    cap->inverterDirecao();
            }
        }
    }
}

void Jogador::mover()
{
    xAnterior = x;
    yAnterior = y;

    // Movimento horizontal pelo teclado
    if (sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Right))
        x += velocidade;

    if (sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Left))
        x -= velocidade;


    x += velocidadeX;
    velocidadeX *= 0.85f;
    if (std::abs(velocidadeX) < 0.2f)
        velocidadeX = 0.f;


    bool cima = sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Up);

    if (cima && !teclaCimaAnterior && noChao)
    {
        velocidadeY = forcaPulo;
    }

    teclaCimaAnterior = cima;


    noChao = false;
    velocidadeY += gravidade;
    y += velocidadeY;


    retangulo->setPosition({x, y});
}

void Jogo::executar()
{
    while (pGG.getJanela().isOpen())
    {
        while (const auto evento = pGG.getJanela().pollEvent())
        {
            if (evento->is<sf::Event::Closed>())
                pGG.getJanela().close();
        }

        // Controle de borda da tecla ESC para n„o alternar estados v·rias vezes seguidas
        bool teclaEsc = sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Escape);
        bool escPressionado = (teclaEsc && !teclaEscAnterior);
        teclaEscAnterior = teclaEsc;

        pGG.getJanela().clear(sf::Color::Black);

        if (estadoAtual == Estado::MENU)
        {
            menu.executar();
            menu.desenhar();

            if (menu.foiConfirmado())
            {
                int opcao = menu.getOpcaoSelecionada();
                menu.resetarConfirmacao();

                if (opcao == 0 || opcao == 1)
                {
                    iniciarFaseEscolhida(opcao);
                }
                else if (opcao == 2)
                {
                    carregarJogoSalvo();
                }
                else if (opcao == 3)
                {
                    estadoAtual = Estado::RANKING;
                }
                else if (opcao == 4)
                {
                    pGG.getJanela().close();
                }
            }
        }
        else if (estadoAtual == Estado::RANKING)
        {
            ranking.desenhar(pGG);

            if (escPressionado)
                estadoAtual = Estado::MENU;
        }
        else if (estadoAtual == Estado::JOGANDO)
        {
            if (faseAtual != nullptr)
            {
                // Apertar ESC pausa o jogo
                if (escPressionado)
                {
                    menuPausa.resetarEstado();
                    mensagemPausa = "";
                    estadoAtual = Estado::PAUSADO;
                }
                else
                {
                    // Atalho r·pido (S) continua funcionando se quiser
                    bool teclaS = sf::Keyboard::isKeyPressed(sf::Keyboard::Key::S);
                    if (teclaS && !teclaSalvarAnterior)
                    {
                        faseAtual->salvar();
                        std::cout << "[JOGO] Jogo salvo!\n";
                    }
                    teclaSalvarAnterior = teclaS;

                    faseAtual->executar();

                    if (faseAtual->estaConcluida() && faseAtual->getNumeroFase() == 1)
                    {
                        delete faseAtual;
                        faseAtual = new Fase_Boss(&pJog1);
                        faseAtual->iniciarFase();
                    }
                    else if ((faseAtual->estaConcluida() && faseAtual->getNumeroFase() == 2) ||
                             !pJog1.estaVivo())
                    {
                        ranking.adicionarPontuacao("Heroi", pJog1.getPontos());
                        delete faseAtual;
                        faseAtual = nullptr;
                        menu.resetarEstado();
                        estadoAtual = Estado::MENU;
                    }
                }
            }
        }
        else if (estadoAtual == Estado::PAUSADO)
        {
            // MantÈm a fase congelada desenhada ao fundo
            if (faseAtual != nullptr)
                faseAtual->desenhar();

            // Fundo escuro semitransparente por cima da fase
            sf::RectangleShape fundoPausa({800.f, 600.f});
            fundoPausa.setFillColor(sf::Color(0, 0, 0, 175));
            pGG.getJanela().draw(fundoPausa);

            // Apertar ESC novamente volta direto para o jogo
            if (escPressionado)
            {
                estadoAtual = Estado::JOGANDO;
            }
            else
            {
                menuPausa.executar();
                menuPausa.desenhar();

                if (!mensagemPausa.empty())
                    pGG.desenharTexto(mensagemPausa, 260.f, 430.f, 22);

                if (menuPausa.foiConfirmado())
                {
                    int opcao = menuPausa.getOpcaoSelecionada();
                    menuPausa.resetarConfirmacao();

                    if (opcao == 0) // Continuar
                    {
                        estadoAtual = Estado::JOGANDO;
                    }
                    else if (opcao == 1) // Salvar Jogo
                    {
                        if (faseAtual != nullptr)
                        {
                            faseAtual->salvar();
                            mensagemPausa = "Jogo salvo com sucesso!";
                        }
                    }
                    else if (opcao == 2) // Voltar ao Menu Principal
                    {
                        delete faseAtual;
                        faseAtual = nullptr;
                        menu.resetarEstado();
                        estadoAtual = Estado::MENU;
                    }
                }
            }
        }

        pGG.getJanela().display();
    }
}

void Menu::resetarEstado()
{
    opcaoSelecionada = 0;
    opcaoConfirmada = false;
    teclaCimaAnterior = true;
    teclaBaixoAnterior = true;
    teclaEnterAnterior = true;
}

void Menu::executar(int comando)
{
    if (comando == 1) // Cima
    {
        opcaoSelecionada--;
        if (opcaoSelecionada < 0)
            opcaoSelecionada = numeroOpcoes - 1;
    }
    else if (comando == 2) // Baixo
    {
        opcaoSelecionada++;
        if (opcaoSelecionada >= numeroOpcoes)
            opcaoSelecionada = 0;
    }
}

void Menu::executar()
{
    bool cima  = sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Up);
    bool baixo = sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Down);
    bool enter = sf::Keyboard::isKeyPressed(sf::Keyboard::Key::Enter);

    if (cima && !teclaCimaAnterior)
        executar(1);

    if (baixo && !teclaBaixoAnterior)
        executar(2);

    if (enter && !teclaEnterAnterior)
        opcaoConfirmada = true;

    teclaCimaAnterior  = cima;
    teclaBaixoAnterior = baixo;
    teclaEnterAnterior = enter;
}

    pGG->desenharTexto(titulo, 270.f, 90.f, 40);

    for (int i = 0; i < numeroOpcoes; ++i)
    {
        std::string linha = (i == opcaoSelecionada)
            ? "> " + opcoes[i]
            : "  " + opcoes[i];

        pGG->desenharTexto(linha, 230.f, 220.f + i * 55.f, 26);
    }

    pGG->desenharTexto("Use Setas (Cima/Baixo) e ENTER", 215.f, 520.f, 18);
}

    }
}

void Planta_Acida::executar()

    if (!estaVivo() && pAcido->estaAtivo()){
        pAcido->executar();
        return;
    }
    else if(!estaVivo())
        return;
    // O proj√©til continua se movimentando mesmo depois de ser disparado.
    pAcido->executar();

    if (alvo == nullptr || pAcido->estaAtivo())
        return;

    // Conta quadros: 90 quadros equivalem a cerca de 1,5 s em 60 FPS.
    ++contadorTempo;
    if (contadorTempo < tempoRecarga)
        return;

    // Compara os centros para decidir para qual lado atirar.
    const float centroPlanta = x + retangulo->getSize().x / 2.f;
    const float centroJogador = alvo->getX() + 20.f;
    const float direcao = (centroJogador >= centroPlanta) ? 1.f : -1.f;

    const float larguraProjetil = 14.f;
    const float inicioX = (direcao > 0.f) ? x + retangulo->getSize().x : x - larguraProjetil;
    const float inicioY = y + 20.f;

    pAcido->ativar(inicioX, inicioY, direcao);
    contadorTempo = 0;
}

void Plataforma::executar()
{
     deslocamentoY = 0.f;

    if (movel)
    {
        const float yTopo = yBase - amplitude;
        float novoY = y + direcao * velocidadeMov;

        if (novoY <= yTopo)       { novoY = yTopo; direcao = 1;  }   // chegou no alto: desce
        else if (novoY >= yBase)  { novoY = yBase; direcao = -1; }   // chegou embaixo: sobe

        deslocamentoY = novoY - y;
        setPosicao(x, novoY);

        // Leva junto o jogador que est· em cima
        if (jogadorApoiado != nullptr)
            jogadorApoiado->setY(jogadorApoiado->getY() + deslocamentoY);
    }

    jogadorApoiado = nullptr;
}

   salvarDataBuffer();
    Salvamento << " " << largura << " " << altura
               << " " << movel << " " << yBase << " " << amplitude
               << " " << velocidadeMov << " " << direcao;
}

void Plataforma::obstaculizar(Jogador* pJogador)
{
    sf::FloatRect jogador = pJogador->getForma()->getGlobalBounds();
    sf::FloatRect plataforma = this->getForma()->getGlobalBounds();

    const float esquerdaJ = jogador.position.x;
    const float direitaJ = jogador.position.x + jogador.size.x;

    const float topoJ = jogador.position.y;
    const float baixoJ = jogador.position.y + jogador.size.y;

    const float esquerdaO = plataforma.position.x;
    const float direitaO = plataforma.position.x + plataforma.size.x;

    const float topoO = plataforma.position.y;
    const float baixoO = plataforma.position.y + plataforma.size.y;

    // Verifica se existe sobreposiÁ„o horizontal
    const bool sobrepoeX =
        direitaJ > esquerdaO &&
        esquerdaJ < direitaO;

    // Colis„o por cima: jogador est· caindo
    // (toler‚ncia: a plataforma pode ter subido ao encontro do jogador neste frame)
    const float tolerancia = 1.f + std::abs(deslocamentoY);

    if (sobrepoeX &&
        pJogador->getVelocidadeY() > 0.f &&
        pJogador->getYAnterior() + jogador.size.y <= topoO + tolerancia &&
        baixoJ >= topoO)
    {
        pJogador->setY(topoO - jogador.size.y);
        pJogador->pararQueda();
        jogadorApoiado = pJogador;      // <-- NOVO: ser· levado no prÛximo frame
    }

    // Colis„o por baixo: jogador est· subindo
    else if (sobrepoeX &&
             pJogador->getVelocidadeY() < 0.f &&
             pJogador->getYAnterior() >= baixoO &&
             topoJ <= baixoO)
    {
        pJogador->setY(baixoO);
        pJogador->pararSubida();
    }

    // Atualiza os limites apÛs possÌveis correÁıes
    jogador = pJogador->getForma()->getGlobalBounds();

    const float esquerdaAtual = jogador.position.x;
    const float direitaAtual = jogador.position.x + jogador.size.x;

    const float topoAtual = jogador.position.y;
    const float baixoAtual = jogador.position.y + jogador.size.y;

    const bool sobrepoeY =
        baixoAtual > topoO &&
        topoAtual < baixoO;

    // Colis„o pela esquerda
    if (sobrepoeY &&
        pJogador->xAnterior + jogador.size.x <= esquerdaO &&
        direitaAtual >= esquerdaO)
    {
        pJogador->setX(esquerdaO - jogador.size.x);
    }

    // Colis„o pela direita
    else if (sobrepoeY &&
             pJogador->xAnterior >= direitaO &&
             esquerdaAtual <= direitaO)
    {
        pJogador->setX(direitaO);
    }
}

#include "Ranking.h"
#include <filesystem>

Ranking::Ranking(const std::string& arquivo)
    : caminhoArquivo(arquivo)
{
    carregarArquivo();
}

void Ranking::carregarArquivo()
{
    registros.clear();
    std::ifstream arq(caminhoArquivo);
    if (!arq.is_open())
        return;

    std::string nome;
    int pontos;
    while (arq >> nome >> pontos)
    {
        registros.push_back({nome, pontos});
    }

    std::sort(registros.begin(), registros.end(),
              [](const Registro& a, const Registro& b) { return a.pontos > b.pontos; });
}

void Ranking::salvarArquivo() const
{
    std::filesystem::create_directories("saves");
    std::ofstream arq(caminhoArquivo);
    if (!arq.is_open())
        return;

    for (const auto& r : registros)
        arq << r.nome << " " << r.pontos << "\n";
}

void Ranking::adicionarPontuacao(const std::string& nome, int pontos)
{
    if (pontos <= 0)
        return;

    carregarArquivo();
    registros.push_back({nome.empty() ? "Heroi" : nome, pontos});

    std::sort(registros.begin(), registros.end(),
              [](const Registro& a, const Registro& b) { return a.pontos > b.pontos; });

    if (registros.size() > 5)
        registros.resize(5);

    salvarArquivo();
}

void Ranking::desenhar(GerenciadorGrafico& pGG) const
{
    pGG.desenharTexto("RANKING - TOP 5", 265.f, 90.f, 34);

    if (registros.empty())
    {
        pGG.desenharTexto("Nenhuma pontuacao registrada ainda.", 210.f, 250.f, 22);
    }
    else
    {
        for (size_t i = 0; i < registros.size(); ++i)
        {
            std::string linha = std::to_string(i + 1) + ". "
                              + registros[i].nome + " - "
                              + std::to_string(registros[i].pontos) + " pts";

            pGG.desenharTexto(linha, 240.f, 190.f + i * 50.f, 24);
        }
    }

    pGG.desenharTexto("Pressione ESC para voltar", 255.f, 510.f, 20);
}
